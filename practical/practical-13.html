02Langkah 1 — Proyeksi pinhole dan disparitas stereo

def proyeksi_pinhole(X, Y, Z, f):
    return f * X / Z, f * Y / Z

x, y = proyeksi_pinhole(2, 1, 10, f=500)
assert (x, y) == (100.0, 50.0)

def depth_dari_disparitas(f, B, d):
    return (f * B) / d

assert abs(depth_dari_disparitas(500, 0.1, 5) - 10.0) < 1e-6
assert abs(depth_dari_disparitas(500, 0.1, 50) - 1.0) < 1e-6
print("Cocok dengan hitungan manual Materi 13.")

03Langkah 2 — Volume rendering NeRF sederhana

import numpy as np

def volume_render(sigma, warna, delta=1.0):
    sigma = np.array(sigma); warna = np.array(warna)
    alpha = 1 - np.exp(-sigma * delta)
    T = np.exp(-np.cumsum(sigma * delta) + sigma * delta)  # transmitansi sebelum titik i
    bobot = T * alpha
    C = (bobot[:, None] * warna).sum(axis=0)
    return C, bobot

sigma = [0.1, 0.5, 0.2]
warna = [[1, 0, 0], [0, 1, 0], [0, 0, 1]]   # merah, hijau, biru

C, bobot = volume_render(sigma, warna)
print("Bobot:", np.round(bobot, 3))     # harus ≈ [0.095, 0.356, 0.099]
print("Warna akhir:", np.round(C, 3))   # harus ≈ [0.095, 0.356, 0.099]

04Langkah 3 — Depth estimation monokuler pretrained

from transformers import DPTImageProcessor, DPTForDepthEstimation
from PIL import Image
import torch, matplotlib.pyplot as plt

processor = DPTImageProcessor.from_pretrained("Intel/dpt-hybrid-midas")
model = DPTForDepthEstimation.from_pretrained("Intel/dpt-hybrid-midas")

img = Image.open("foto_sendiri.jpg")
inputs = processor(images=img, return_tensors="pt")
with torch.no_grad():
    depth = model(**inputs).predicted_depth

depth = torch.nn.functional.interpolate(
    depth.unsqueeze(1), size=img.size[::-1], mode="bicubic"
).squeeze().numpy()

plt.subplot(1, 2, 1); plt.imshow(img); plt.title("Asli"); plt.axis("off")
plt.subplot(1, 2, 2); plt.imshow(depth, cmap="viridis"); plt.title("Depth (relatif)"); plt.axis("off")
plt.show()

Ulangi pada lima foto dengan objek di jarak berbeda. Urutkan objek dari yang menurut Anda paling dekat sampai paling jauh, lalu bandingkan dengan urutan nilai depth dari model.
