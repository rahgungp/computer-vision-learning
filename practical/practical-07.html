02Langkah 1 — Self-attention dari nol

import numpy as np

def softmax(x):
    e = np.exp(x - np.max(x, axis=-1, keepdims=True))   # stabil numerik
    return e / e.sum(axis=-1, keepdims=True)

def self_attention(X, Wq, Wk, Wv):
    Q, K, V = X @ Wq, X @ Wk, X @ Wv
    d_k = K.shape[-1]
    skor = (Q @ K.T) / np.sqrt(d_k)
    bobot = softmax(skor)
    return bobot @ V, bobot

X = np.array([[1, 0], [0, 1]], dtype=float)     # dua token, dim 2
I = np.eye(2)                                   # Wq = Wk = Wv = identitas

keluar, bobot = self_attention(X, I, I, I)
print("Bobot attention:\n", bobot)
print("Output:\n", keluar)

harapan = np.array([[0.670, 0.330], [0.330, 0.670]])
assert np.allclose(keluar, harapan, atol=1e-3), "belum cocok - periksa kembali implementasi Anda"
print("Cocok dengan hitungan manual Materi 07.")
