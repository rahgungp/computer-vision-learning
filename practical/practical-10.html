VAE dari Nol & Diffusion Pretrained

Reparameterization trick, sampling citra baru, dan eksperimen jumlah langkah denoising

⏱ 120 menit 🧰 Google Colab (GPU) · PyTorch · Diffusers 📖 Materi 10

01Tujuan Praktikum

Setelah praktikum ini, mahasiswa mampu:

  • mengimplementasikan VAE sederhana dari nol, lengkap dengan reparameterization trick;
  • menghitung KL divergence dan reconstruction loss secara terpisah lalu menggabungkannya;
  • men-sample citra baru dari ruang laten dan menilai kualitasnya secara visual;
  • menjalankan pipeline diffusion pretrained dan membandingkan trade-off jumlah langkah inferensi.

02Langkah 1 — VAE kecil pada MNIST

import torch
import torch.nn as nn
import torch.nn.functional as F

class VAE(nn.Module):
    def __init__(self, dim_laten=2):
        super().__init__()
        self.enc = nn.Sequential(nn.Linear(784, 256), nn.ReLU())
        self.fc_mu     = nn.Linear(256, dim_laten)
        self.fc_logvar = nn.Linear(256, dim_laten)
        self.dec = nn.Sequential(
            nn.Linear(dim_laten, 256), nn.ReLU(),
            nn.Linear(256, 784), nn.Sigmoid()
        )

    def encode(self, x):
        h = self.enc(x)
        return self.fc_mu(h), self.fc_logvar(h)

    def reparameterize(self, mu, logvar):
        sigma = torch.exp(0.5 * logvar)
        eps = torch.randn_like(sigma)
        return mu + sigma * eps          # z = mu + sigma ⊙ eps

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        return self.dec(z), mu, logvar

03Langkah 2 — Loss gabungan dan verifikasi KL manual

def loss_vae(rekonstruksi, x, mu, logvar):
    recon = F.binary_cross_entropy(rekonstruksi, x, reduction="sum")
    kl = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp())
    return recon + kl, recon, kl

# Verifikasi terhadap hitungan manual Materi 10
mu, logvar = torch.tensor([2.0]), torch.log(torch.tensor([3.0]))
kl_manual = -0.5 * (1 + logvar - mu**2 - logvar.exp())
print("KL (mu=2, sigma^2=3):", kl_manual.item())   # harus ≈ 2.451

04Langkah 3 — Latih dan sample citra baru

model = VAE(dim_laten=2)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

for epoch in range(20):
    for x, _ in train_loader:
        x = x.view(-1, 784)
        rekon, mu, logvar = model(x)
        loss, recon, kl = loss_vae(rekon, x, mu, logvar)
        optimizer.zero_grad(); loss.backward(); optimizer.step()
    print(f"epoch {epoch}: loss={loss.item():.1f}  recon={recon.item():.1f}  kl={kl.item():.1f}")

# Sample 10 citra baru langsung dari N(0, I) -- tanpa citra input sama sekali

05Langkah 4 — Diffusion pretrained: trade-off langkah inferensi

from diffusers import DDPMPipeline, DDIMScheduler
import time

pipeline = DDPMPipeline.from_pretrained("google/ddpm-celebahq-256")
pipeline.scheduler = DDIMScheduler.from_config(pipeline.scheduler.config)

for n_step in [10, 50, 200, 1000]:
    mulai = time.time()
    img = pipeline(num_inference_steps=n_step).images[0]
    durasi = time.time() - mulai
    img.save(f"hasil_{n_step}langkah.png")
    print(f"{n_step:>5} langkah  →  {durasi:6.1f} detik")
