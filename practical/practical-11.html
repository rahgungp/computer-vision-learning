02Langkah 1 — IoU dari nol

def iou(boxA, boxB):
    xA = max(boxA[0], boxB[0]); yA = max(boxA[1], boxB[1])
    xB = min(boxA[2], boxB[2]); yB = min(boxA[3], boxB[3])
    inter = max(0, xB - xA) * max(0, yB - yA)
    areaA = (boxA[2]-boxA[0]) * (boxA[3]-boxA[1])
    areaB = (boxB[2]-boxB[0]) * (boxB[3]-boxB[1])
    return inter / (areaA + areaB - inter)

gt = (50, 40, 150, 140)
assert abs(iou(gt, (90, 60, 190, 160)) - 0.316) < 1e-2   # Kasus A
assert abs(iou(gt, (60, 50, 160, 150)) - 0.681) < 1e-2   # Kasus B
print("Cocok dengan hitungan manual Materi 11.")

03Langkah 2 — Average Precision dari nol

def average_precision(status_terurut, total_gt):
    tp = fp = 0
    precisions, recalls = [], []
    for status in status_terurut:
        tp += status == "TP"; fp += status == "FP"
        precisions.append(tp / (tp + fp))
        recalls.append(tp / total_gt)

    ap, prev_r = 0.0, 0.0
    for r, p in zip(recalls, precisions):
        if r > prev_r:
            p_interp = max(pp for pp, rr in zip(precisions, recalls) if rr >= prev_r)
            ap += (r - prev_r) * p_interp
            prev_r = r
    return ap

contoh = ["TP", "TP", "FP", "TP", "FP"]
ap = average_precision(contoh, total_gt=5)
print(round(ap, 3))                 # harus ≈ 0.55
assert abs(ap - 0.55) < 1e-2
print("Cocok dengan hitungan manual Materi 11.")

05Langkah 4 — mIoU untuk segmentasi

import numpy as np

def miou(cm):            # cm = confusion matrix, cm[i, j] = pixel aktual kelas i diprediksi kelas j
    ious = []
    for k in range(len(cm)):
        tp = cm[k, k]
        fp = cm[:, k].sum() - tp
        fn = cm[k, :].sum() - tp
        ious.append(tp / (tp + fp + fn))
    return np.mean(ious), ious

cm = np.array([[800, 200],
               [100, 900]])
rata, per_kelas = miou(cm)
print("IoU per kelas:", per_kelas)   # harus ≈ [0.727, 0.75]
print("mIoU:", round(rata, 3))       # harus ≈ 0.739
